async page => {
  const assert = (condition, message) => { if (!condition) throw new Error(message) }
  await page.emulateMedia({reducedMotion:'no-preference'})
  await page.setViewportSize({width:1440,height:1000})
  await page.goto('http://localhost:3001')
  await page.locator('#uslugi').waitFor()
  await page.getByRole('button', {name:/Chcę sprawniej obsługiwać/}).click()
  await page.getByRole('button', {name:'Powtórz pokaz',exact:true}).click()
  await page.getByRole('button', {name:'Zatrzymaj pokaz',exact:true}).click()
  const times = () => page.locator('[data-scene-motion]').evaluateAll(elements => elements.flatMap(e => e.getAnimations().map(a => Number(a.currentTime))))
  const before = await times()
  await page.waitForTimeout(250)
  assert(JSON.stringify(before) === JSON.stringify(await times()), 'Pause must stop every scene animation')
  await page.getByRole('button', {name:'Wznów pokaz',exact:true}).click()
  await page.waitForTimeout(250)
  assert((await times())[0] > before[0], 'Resume must advance timeline')
  await page.getByRole('button', {name:'Pokaż rezultat',exact:true}).click()
  assert((await times()).every(time => time === 7000), 'Skip must finish all scene layers')
  await page.getByRole('button', {name:'Powtórz pokaz',exact:true}).click()
  assert((await times()).every(time => time < 1500), 'Replay must reset all layers')
  await page.evaluate(() => scrollTo(0,0))
  await page.waitForTimeout(150)
  assert(await page.locator('[data-scene-motion]').evaluateAll(elements => elements.every(e => e.getAnimations().every(a => a.playState === 'paused'))), 'Offscreen scenes must pause')
  await page.getByRole('button', {name:/Mam inny pomysł/}).click()
  await page.locator('#scenario-idea').fill('a'.repeat(2001))
  assert((await page.locator('#scenario-idea').inputValue()).length === 2001, 'Pasted description must not be silently truncated')
  await page.getByRole('button', {name:/Napisz do nas/}).click()
  assert(await page.locator('#scenario-idea-error').isVisible(), 'Overlong description must be rejected')
  await page.locator('#scenario-idea').fill('Chcę przetestować obsługę szkicu bez wysyłania.')
  await page.getByRole('button', {name:/Przejdź do czatu/}).click()
  await page.keyboard.press('Escape')
  assert(await page.getByRole('button', {name:/Przejdź do czatu/}).evaluate(e => e === document.activeElement), 'Closing chat must restore focus')
  return 'PASS: pause/resume/skip/replay, offscreen pause, overlong draft preserved and rejected, focus return'
}
