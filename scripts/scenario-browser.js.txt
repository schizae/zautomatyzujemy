async page => {
  const assert = (condition, message) => { if (!condition) throw new Error(message) }
  const requests = []
  page.on('request', request => { if (request.method() === 'POST') requests.push(request.url()) })
  await page.goto('http://localhost:3001')
  await page.waitForFunction(() => document.querySelectorAll('#uslugi').length === 1 && document.querySelector('#uslugi')?.checkVisibility() && !document.getElementById('S:0'))
  const section = page.locator('#uslugi')
  await section.waitFor({state:'visible',timeout:30000})
  const cookie = page.getByRole('button', {name:'Rozumiem',exact:true})
  if (await cookie.isVisible()) await cookie.click()
  await page.setViewportSize({width:1440,height:1000})
  await section.scrollIntoViewIfNeeded()
  const finish = page.getByRole('button', {name:'Pokaż rezultat',exact:true})
  if (await finish.isEnabled()) await finish.click()
  await section.screenshot({path:'.playwright-cli/scenarios-desktop.png'})
  await page.getByRole('button', {name:/Mam inny pomysł/}).click()
  await page.getByRole('button', {name:/Przejdź do czatu/}).click()
  assert(await page.locator('#scenario-idea-error').isVisible(), 'Empty idea must be rejected')
  const idea = 'Chcę aplikację do zarządzania rezerwacjami mojego zespołu.'
  await page.locator('#scenario-idea').fill(idea)
  await page.getByRole('button', {name:/Przejdź do czatu/}).click()
  await page.getByRole('textbox', {name:'Wiadomość do Klary',exact:true}).waitFor()
  assert(await page.getByRole('textbox', {name:'Wiadomość do Klary',exact:true}).inputValue() === idea, 'Chat must receive the draft')
  await page.getByRole('textbox', {name:'Wiadomość do Klary',exact:true}).fill('Mój wcześniejszy niewysłany szkic')
  await page.keyboard.press('Escape')
  await page.getByRole('button', {name:/Przejdź do czatu/}).click()
  assert(await page.getByRole('textbox', {name:'Wiadomość do Klary',exact:true}).inputValue() === 'Mój wcześniejszy niewysłany szkic', 'Chat must not overwrite existing draft')
  assert(await page.getByText('Masz już niewysłaną wiadomość.', {exact:false}).count() > 0 || await page.getByRole('button', {name:/Zastąp/}).count() > 0, 'Conflict preview must be offered')
  await page.keyboard.press('Escape')
  await page.getByRole('button', {name:/Napisz do nas/}).click()
  assert(await page.locator('#scenario-email-message').inputValue() === idea, 'Email must receive the draft')
  await page.locator('#scenario-email-name').fill('Testowy Klient')
  await page.locator('#scenario-email-email').fill('preview@example.com')
  await page.getByRole('button', {name:/Chcę więcej zapytań/}).click()
  await page.getByRole('button', {name:/Mam inny pomysł/}).click()
  assert(await page.locator('#scenario-email-name').inputValue() === 'Testowy Klient', 'Name must survive scenario change')
  assert(await page.locator('#scenario-email-email').inputValue() === 'preview@example.com', 'Email must survive scenario change')
  assert(await page.locator('#scenario-email-message').inputValue() === idea, 'Idea must survive scenario change')
  await section.screenshot({path:'.playwright-cli/scenarios-email.png'})
  for (const width of [375,768,1440]) {
    await page.setViewportSize({width,height:1000})
    await page.waitForFunction(() => document.documentElement.scrollWidth <= innerWidth + 1)
    assert(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth + 1), `Overflow at ${width}`)
  }
  await page.setViewportSize({width:375,height:900})
  await page.locator('#service-detail').screenshot({path:'.playwright-cli/scenarios-mobile.png'})
  await page.emulateMedia({reducedMotion:'reduce'})
  await page.getByRole('button', {name:/Chcę odzyskać czas/}).click()
  assert(await page.getByTestId('scene-stage').getAttribute('data-stage') === '2', 'Reduced motion must show final result')
  assert(await page.getByRole('button', {name:'Powtórz pokaz',exact:true}).isDisabled(), 'Reduced motion must disable replay')
  const uniqueIds = await page.evaluate(() => { const ids = [...document.querySelectorAll('[id]')].map(element => element.id); return ids.length === new Set(ids).size })
  assert(uniqueIds, 'IDs must be unique')
  assert(!requests.some(url => url.includes('/api/chat') || url.includes('glos')), 'Opening paths must not send to chat or voice')
  return ('PASS: validation, chat handoff/conflict, email persistence, 3 widths, reduced motion, unique IDs, no automatic conversation sends')
}
